-- Tablas para el sistema de paquetes dinamicos
CREATE TABLE IF NOT EXISTS pricing_packages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    created_at TIMESTAMPTZ DEFAULT now(),
    client_name TEXT NOT NULL,
    services JSONB NOT NULL,
    total_setup NUMERIC NOT NULL,
    total_monthly NUMERIC NOT NULL,
    image_url TEXT,
    package_type TEXT
);

-- Habilitar RLS
ALTER TABLE pricing_packages ENABLE ROW LEVEL SECURITY;

-- Politicas de acceso publico (Permitir insercion y lectura anonima si asi lo requiere el flujo actual)
DROP POLICY IF EXISTS "Permitir insercion publica" ON pricing_packages;
CREATE POLICY "Permitir insercion publica" ON pricing_packages FOR INSERT WITH CHECK (true);

DROP POLICY IF EXISTS "Permitir lectura publica" ON pricing_packages;
CREATE POLICY "Permitir lectura publica" ON pricing_packages FOR SELECT USING (true);

NOTIFY pgrst, 'reload schema';
